import sqlite3
import math
import json
import os

# Database Path
DB_PATH = "/workspace/scratch/hyper_rag_trading.db"

def init_db(db_path=DB_PATH):
    os.makedirs(os.path.dirname(db_path), exist_ok=True)
    conn = sqlite3.connect(db_path)
    cursor = conn.cursor()
    
    # 1. Hypergraph Nodes Table
    cursor.execute("""
    CREATE TABLE IF NOT EXISTS hypergraph_nodes (
        node_id TEXT PRIMARY KEY,
        node_type TEXT NOT NULL,
        name TEXT NOT NULL,
        metadata_json TEXT
    );
    """)
    
    # 2. Hyperedges Table
    cursor.execute("""
    CREATE TABLE IF NOT EXISTS hyperedges (
        edge_id TEXT PRIMARY KEY,
        edge_type TEXT NOT NULL,
        source_nodes TEXT NOT NULL,
        target_nodes TEXT NOT NULL,
        weight REAL DEFAULT 1.0,
        relational_dialogue TEXT
    );
    """)
    
    # 3. Trade Setups Table
    cursor.execute("""
    CREATE TABLE IF NOT EXISTS trade_setups (
        ticker TEXT PRIMARY KEY,
        sector TEXT NOT NULL,
        pivot_breakout REAL NOT NULL,
        target_price REAL NOT NULL,
        stop_loss REAL NOT NULL,
        catalyst TEXT NOT NULL,
        win_rate REAL NOT NULL,
        risk_reward_odds REAL NOT NULL,
        expected_return_pct REAL NOT NULL,
        variance_drag_pct REAL NOT NULL,
        net_edge_pct REAL NOT NULL,
        ruin_prob_pct REAL NOT NULL,
        kelly_fraction REAL NOT NULL,
        allocated_capital REAL NOT NULL,
        gate1_ruin_pass INTEGER NOT NULL,
        gate2_ergodicity_pass INTEGER NOT NULL,
        gate3_edge_pass INTEGER NOT NULL
    );
    """)
    conn.commit()
    return conn

def calculate_kelly_and_gates(entry, target, stop, win_rate=0.62, cap_shield=0.5, capital=1008.0):
    """
    Calculates Risk-Reward (b), Fractional Kelly Sizing, Variance Drag, Net Edge,
    Ruin Probability, and evaluates the 3-Gate Variance Shield.
    Equations grounded in Notebook Topics 001, 089, 090, 092 [4, 10, 12].
    """
    risk = entry - stop
    reward = target - entry
    if risk <= 0 or reward <= 0:
        raise ValueError("Invalid trade geometry: Risk and Reward must be positive.")
    
    b = reward / risk  # Odds
    p = win_rate
    q = 1.0 - p
    
    # Full Kelly: f* = (p*b - q) / b; Fractional Kelly = f* * CapShield [4, 10]
    full_kelly = (b * p - q) / b
    fractional_kelly = max(0.0, full_kelly * cap_shield)
    
    # Expected Return (mu) & Volatility (sigma) per trade
    r_win = reward / entry
    r_loss = -risk / entry
    mu = p * r_win + q * r_loss
    sigma = math.sqrt(p * (r_win - mu)**2 + q * (r_loss - mu)**2)
    
    # Variance Drag = sigma^2 / 2
    variance_drag = (sigma**2) / 2.0
    net_edge = mu - variance_drag
    
    # Ruin Probability Approximation: P(Ruin) = exp(-2 * mu / sigma^2) [12, 26]
    ruin_prob = math.exp(-2.0 * max(1e-5, mu) / (sigma**2 + 1e-6))
    
    # 3-Gate Variance Shield Evaluation
    gate1 = 1 if ruin_prob < 0.015 else 0      # Gate 1: Ruin Prob < 1.5%
    gate2 = 1 if net_edge >= 0.0 else 0        # Gate 2: Ergodicity Survival >= 90%
    gate3 = 1 if mu > variance_drag else 0      # Gate 3: Expected Edge > Variance Drag
    
    allocated_capital = capital * fractional_kelly
    
    return {
        "odds_b": round(b, 2),
        "mu_pct": round(mu * 100, 2),
        "variance_drag_pct": round(variance_drag * 100, 4),
        "net_edge_pct": round(net_edge * 100, 2),
        "ruin_prob_pct": round(ruin_prob * 100, 2),
        "kelly_fraction": round(fractional_kelly, 4),
        "allocated_capital": round(allocated_capital, 2),
        "gate1": gate1,
        "gate2": gate2,
        "gate3": gate3
    }

def execute_dhanhq_v2_order(ticker, pivot, qty, gate_pass, seconds_since_open=75):
    """
    Simulates order execution via DhanHQ API v2.
    Grounded in Broker API Integrations & 65s Opening Wick Quarantine [3, 25, 28].
    """
    if seconds_since_open < 65:
        return {"status": "REJECTED", "reason": "Opening Wick Quarantine active (<65s post-open)"}
    if not gate_pass:
        return {"status": "REJECTED", "reason": "Failed 3-Gate Variance Shield"}
    
    payload = {
        "dhanClientId": "1000000001",
        "transactionType": "BUY",
        "exchangeSegment": "NSE_EQ",
        "productType": "INTRADAY",
        "orderType": "STOP_LOSS_MARKET",
        "securityId": ticker,
        "quantity": int(qty),
        "price": 0.0,
        "triggerPrice": pivot,
        "validity": "DAY"
    }
    return {"status": "SUCCESS", "order_id": f"DHAN_{ticker}_99821", "payload": payload}
