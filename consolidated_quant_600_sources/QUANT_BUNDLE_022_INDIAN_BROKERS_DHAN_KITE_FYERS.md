"""
setup.py — One-time tool to look up Dhan security IDs for your watchlist symbols
and verify the mapping in dhan_client.py is correct.

Run once before going live:
    python setup.py
"""

import sys

from dhanhq import dhanhq, DhanContext

import config
from dhan_client import SYMBOL_TO_SECURITY_ID


def main():
    ctx = DhanContext(config.DHAN_CLIENT_ID, config.DHAN_ACCESS_TOKEN)
    client = dhanhq(ctx)
    print(f"\nWatchlist: {config.WATCHLIST}\n")
    print(f"{'Symbol':<15} {'Mapped ID':<15} {'Exchange':<10} {'LTP'}")
    print("-" * 55)

    all_ok = True
    for symbol in config.WATCHLIST:
        mapped_id = SYMBOL_TO_SECURITY_ID.get(symbol, "NOT MAPPED")
        if mapped_id == "NOT MAPPED":
            print(f"{symbol:<15} {'NOT MAPPED':<15}")
            all_ok = False
            continue

        try:
            resp = client.ohlc_data({"NSE_EQ": [int(mapped_id)]})
            data = (resp.get("data", {}).get("NSE_EQ") or {})
            entry = data.get(mapped_id) or data.get(int(mapped_id)) or {}
            ltp = entry.get("last_price") or entry.get("ltp", "N/A")
            print(f"{symbol:<15} {mapped_id:<15} {'NSE_EQ':<10} ₹{ltp}")
        except Exception as exc:
            print(f"{symbol:<15} {mapped_id:<15} ERROR: {exc}")
            all_ok = False

    print()
    if all_ok:
        print("✅ All symbols verified. Ready to trade.")
    else:
        print("⚠️  Some symbols failed. Update SYMBOL_TO_SECURITY_ID in dhan_client.py.")
        print("   Search for IDs at: https://images.dhan.co/api-data/api-scrip-master.csv")
        sys.exit(1)

"""
whatsapp_bot.py — WhatsApp notifications + inbound command handler via Twilio

Outbound: push alerts (trade signals, P&L summary) via Twilio REST API.
Inbound:  a lightweight Flask webhook receives commands typed in WhatsApp.
          Supported commands:

            /pnl      — today's P&L
            /help     — show available commands

Run the webhook server in a background thread alongside the trader.
For local testing, expose it with: ngrok http 5001
Then set your Twilio sandbox webhook URL to: https://<ngrok-url>/whatsapp

    resp = MessagingResponse()
    msg = resp.message()

    if _trader_ref is None:
        msg.body("⚠️ Trader not initialised yet.")
        return str(resp)

    if incoming in ("/help", "help"):
        msg.body(
            "*Trading Bot Commands*\n"
            "/status    — positions & P&L\n"
            "/pause     — pause trading\n"
            "/resume    — resume trading\n"
            "/squareoff — close all positions\n"
            "/pnl       — today's P&L\n"
            "/help      — this message"
        )

    elif incoming == "/status":
        positions = _trader_ref.open_positions
        pnl = _trader_ref.dhan.get_daily_pnl()
        if positions:
            lines = [f"*Open Positions ({len(positions)})*"]
            for sym, info in positions.items():

            msg.body("\n".join(lines))
        else:
            msg.body(f"No open positions.\nP&L: ₹{pnl:+.2f}")

    elif incoming == "/pause":
        _trader_ref.paused = True
        msg.body("⏸ Trading paused.")

    elif incoming == "/resume":
        _trader_ref.paused = False
        msg.body("▶️ Trading resumed.")

    elif incoming == "/squareoff":
        count = _trader_ref.dhan.square_off_all()
        _trader_ref.open_positions.clear()
        msg.body(f"⏹ Squared off {count} position(s).")

    elif incoming == "/pnl":
        pnl = _trader_ref.dhan.get_daily_pnl()
        emoji = "✅" if pnl >= 0 else "❌"
        msg.body(f"{emoji} Today's P&L: ₹{pnl:+.2f}")

    else:
        msg.body("Unknown command. Send /help for the list of commands.")

    return str(resp)

def start_webhook_server(trader: "Trader", port: int = 5001) -> None:
    """Start the Flask webhook server in a daemon thread."""

    _trader_ref = trader

    def _run():
        log = logging.getLogger("werkzeug")
        log.setLevel(logging.WARNING)
        app.run(host="0.0.0.0", port=port, debug=False, use_reloader=False)

    thread = threading.Thread(target=_run, daemon=True, name="whatsapp-webhook")
    thread.start()
    logger.info("WhatsApp webhook server started on port %d", port)

"""
gemini_strategy.py — Technical indicator computation + Gemini 1.5 Flash signal engine

Signal schema returned by analyse():
    {
        "signal":     "BUY" | "SELL" | "HOLD",
        "confidence": 0–100,      # Gemini's stated confidence %
        "entry":      float,      # Suggested entry price
        "stop_loss":  float,      # Suggested stop-loss price
        "target":     float,      # Suggested target price
        "reasoning":  str,        # One-line rationale
    }
"""

import json
import logging
import math
from typing import Optional

from google import genai
from google.genai import types

import config

logger = logging.getLogger(__name__)

_client = genai.Client(api_key=config.GEMINI_API_KEY)

# ── Technical indicator helpers ───────────────────────────────────────────────


def _ema(values: list[float], period: int) -> list[float]:
    if len(values) < period:
        return []
    k = 2 / (period + 1)
    result = [sum(values[:period]) / period]
    for v in values[period:]:
        result.append(v * k + result[-1] * (1 - k))
    return result


def _rsi(closes: list[float], period: int = 14) -> Optional[float]:
    if len(closes) < period + 1:
        return None
    gains, losses = [], []
    for i in range(1, len(closes)):
        diff = closes[i] - closes[i - 1]
        gains.append(max(diff, 0))
        losses.append(max(-diff, 0))
    avg_gain = sum(gains[-period:]) / period
    avg_loss = sum(losses[-period:]) / period
    if avg_loss == 0:
        return 100.0
    rs = avg_gain / avg_loss
    return round(100 - (100 / (1 + rs)), 2)


def _macd(closes: list[float]) -> dict:
    ema12 = _ema(closes, 12)
    ema26 = _ema(closes, 26)
    if not ema12 or not ema26:
        return {}
    # Align lengths
    diff = len(ema12) - len(ema26)
    ema12 = ema12[diff:]
    macd_line = [a - b for a, b in zip(ema12, ema26)]
    signal_line = _ema(macd_line, 9)
    if not signal_line:
        return {}
    histogram = macd_line[-len(signal_line):]
    return {
        "macd": round(macd_line[-1], 4),
        "signal": round(signal_line[-1], 4),
        "histogram": round(histogram[-1], 4),
    }


def _bollinger(closes: list[float], period: int = 20, std_mult: float = 2.0) -> dict:
    if len(closes) < period:
        return {}
    window = closes[-period:]
    mean = sum(window) / period
    variance = sum((x - mean) ** 2 for x in window) / period
    std = math.sqrt(variance)
    return {
        "upper": round(mean + std_mult * std, 2),
        "middle": round(mean, 2),
        "lower": round(mean - std_mult * std, 2),
    }


def _atr(candles: list[dict], period: int = 14) -> Optional[float]:
    if len(candles) < period + 1:
        return None
    trs = []
    for i in range(1, len(candles)):
        high = candles[i]["high"]
        low = candles[i]["low"]
        prev_close = candles[i - 1]["close"]
        trs.append(max(high - low, abs(high - prev_close), abs(low - prev_close)))
    return round(sum(trs[-period:]) / period, 2)


def compute_indicators(candles: list[dict]) -> dict:
    """Return dict of all indicators from a list of OHLCV candles."""
    closes = [c["close"] for c in candles]
    ema9 = _ema(closes, 9)
    ema21 = _ema(closes, 21)
    ema50 = _ema(closes, 50)
    return {
        "last_close": closes[-1] if closes else None,
        "rsi_14": _rsi(closes),
        "macd": _macd(closes),
        "ema9": round(ema9[-1], 2) if ema9 else None,
        "ema21": round(ema21[-1], 2) if ema21 else None,
        "ema50": round(ema50[-1], 2) if ema50 else None,
        "bollinger": _bollinger(closes),
        "atr_14": _atr(candles),
        "volume_last": candles[-1]["volume"] if candles else None,
        "volume_avg_20": (
            round(sum(c["volume"] for c in candles[-20:]) / min(20, len(candles)), 0)
            if candles else None
        ),
    }


# ── Gemini prompt & parsing ───────────────────────────────────────────────────

_PROMPT_TEMPLATE = """
You are an expert intraday stock trader for NSE India (NSE equities, MIS intraday).
Analyse the following technical indicators for {symbol} and provide a trading signal.

## Indicators (5-minute candles)
{indicators_json}

## Rules
- Trade only during 9:30 AM – 3:20 PM IST.
- This is INTRADAY — all positions must be squared off by 3:20 PM.
- Use RSI, MACD, EMA crossovers, Bollinger Bands, and ATR for your analysis.
- Only give BUY or SELL signals with confidence >= 60%. Otherwise HOLD.

## Response format (JSON only, no markdown, no extra text)
{{
  "signal": "BUY" | "SELL" | "HOLD",
  "confidence": <integer 0-100>,
  "entry": <float>,
  "stop_loss": <float>,
  "target": <float>,
  "reasoning": "<one line>"
}}
""".strip()


def analyse(symbol: str, candles: list[dict]) -> Optional[dict]:
    """
    Run Gemini strategy on the given candles.
    Returns signal dict or None on failure.
    """
    if len(candles) < 30:
        logger.warning("%s: not enough candles (%d)", symbol, len(candles))
        return None

    indicators = compute_indicators(candles)
    prompt = _PROMPT_TEMPLATE.format(
        symbol=symbol,
        indicators_json=json.dumps(indicators, indent=2),
    )

    try:
        response = _client.models.generate_content(
            model=config.GEMINI_MODEL,
            contents=prompt,
        )
        raw = response.text.strip()
        # Strip markdown fences if present
        if raw.startswith("```"):
            raw = raw.split("```")[1]
            if raw.startswith("json"):
                raw = raw[4:]
        signal = json.loads(raw.strip())
        # Validate required keys
        for key in ("signal", "confidence", "entry", "stop_loss", "target"):
            if key not in signal:
                raise ValueError(f"Missing key: {key}")
        signal["signal"] = signal["signal"].upper()
        signal["confidence"] = int(signal["confidence"])
        return signal
    except Exception as exc:
        logger.error("Gemini analysis for %s failed: %s", symbol, exc)
        return None

"""
trader.py — Main trading loop

Schedule:
  9:30 AM  → scan starts, Gemini analyses each watchlist symbol every POLL_INTERVAL_SEC
  3:20 PM  → auto square-off all open positions
  3:30 PM  → daily P&L summary sent to WhatsApp, loop exits

Mon–Sat only (Dhan supports Saturday sessions on some exchanges).
"""

import asyncio
import logging
import math
from datetime import datetime, time

import config
import gemini_strategy as strategy
import whatsapp_bot as wa
from dhan_client import DhanClient

logger = logging.getLogger(__name__)

MARKET_OPEN = time(config.MARKET_OPEN_H, config.MARKET_OPEN_M)
SQUAREOFF_TIME = time(config.SQUAREOFF_H, config.SQUAREOFF_M)
MARKET_CLOSE = time(config.MARKET_CLOSE_H, config.MARKET_CLOSE_M)


class Trader:
    def __init__(self) -> None:
        self.dhan = DhanClient()
        self.paused: bool = False
        self.open_positions: dict[str, dict] = {}   # symbol → {side, qty, entry, order_id}
        self.trades_today: int = 0
        self._squaredoff: bool = False

    # ── Helpers ───────────────────────────────────────────────────────────────

    @staticmethod
    def _now() -> datetime:
        return datetime.now()

    @staticmethod
    def _market_day() -> bool:
        """True on Mon–Sat."""
        return datetime.now().weekday() < 6  # 0=Mon … 5=Sat

    def _in_trading_window(self) -> bool:
        now_t = self._now().time()
        return MARKET_OPEN <= now_t < SQUAREOFF_TIME

    def _past_squareoff(self) -> bool:
        return self._now().time() >= SQUAREOFF_TIME

    def _calc_qty(self, price: float) -> int:
        """Whole shares that fit within CAPITAL_PER_TRADE."""
        if price <= 0:
            return 0
        return max(1, math.floor(config.CAPITAL_PER_TRADE / price))

    # ── Core logic ────────────────────────────────────────────────────────────

    async def _analyse_and_trade(self, symbol: str) -> None:
        """Fetch candles, run Gemini, place order if signal is strong enough."""
        if symbol in self.open_positions:
            logger.debug("Already in position for %s, skipping", symbol)
            return

        candles = await asyncio.to_thread(self.dhan.get_candles, symbol, "5", 3)
        if not candles:
            logger.warning("No candles for %s", symbol)
            return

        signal = await asyncio.to_thread(strategy.analyse, symbol, candles)
        if signal is None:
            return

        action = signal["signal"]
        confidence = signal["confidence"]

        logger.info(
            "%s → %s (conf=%d%%) entry=%.2f sl=%.2f tgt=%.2f",
            symbol, action, confidence, signal["entry"], signal["stop_loss"], signal["target"],
        )

        if action == "HOLD" or confidence < config.MIN_CONFIDENCE:
            return

        if len(self.open_positions) >= config.MAX_POSITIONS:
            logger.info("Max positions (%d) reached, skipping %s", config.MAX_POSITIONS, symbol)
            return

        ltp = await asyncio.to_thread(self.dhan.get_ltp, symbol)
        qty = self._calc_qty(ltp or signal["entry"])
        if qty == 0:
            return

        order_id = await asyncio.to_thread(
            self.dhan.place_order, symbol, action, qty
        )
        if order_id:
            self.open_positions[symbol] = {
                "side": action,
                "qty": qty,
                "entry": ltp or signal["entry"],
                "order_id": order_id,
            }
            self.trades_today += 1
            wa.send_signal_alert(symbol, signal, order_id)

    async def _squareoff_all(self) -> None:
        if self._squaredoff:
            return
        self._squaredoff = True
        logger.info("3:20 PM — squaring off all positions")
        count = await asyncio.to_thread(self.dhan.square_off_all)
        self.open_positions.clear()
        wa.send_squareoff_notice(count)

    async def _send_eod_summary(self) -> None:
        pnl = await asyncio.to_thread(self.dhan.get_daily_pnl)
        wa.send_pnl_summary(pnl, self.trades_today)

    # ── Main loop ─────────────────────────────────────────────────────────────

    async def run(self) -> None:
        logger.info(
            "Trader started | mode=%s | watchlist=%s",
            config.TRADE_MODE, config.WATCHLIST,
        )
        wa.send(
            f"🤖 *Trading Bot Started*\n"
            f"Mode: {config.TRADE_MODE}\n"
            f"Watchlist: {', '.join(config.WATCHLIST)}\n"
            f"Min confidence: {config.MIN_CONFIDENCE}%"
        )

        eod_sent = False

        while True:
            now = self._now()

            # Only run on trading days
            if not self._market_day():
                logger.info("Non-trading day (%s). Sleeping 1 hour.", now.strftime("%A"))
                await asyncio.sleep(3600)
                continue

            now_t = now.time()

            # Before market open — wait
            if now_t < MARKET_OPEN:
                wait_sec = (
                    datetime.combine(now.date(), MARKET_OPEN) - now
                ).seconds
                logger.info("Market opens in %d min. Waiting…", wait_sec // 60)
                await asyncio.sleep(min(wait_sec, 300))
                continue

            # Auto square-off
            if now_t >= SQUAREOFF_TIME and not self._squaredoff:
                await self._squareoff_all()

            # EOD summary + exit
            if now_t >= MARKET_CLOSE:
                if not eod_sent:
                    await self._send_eod_summary()
                    eod_sent = True
                logger.info("Market closed. Trader shutting down for today.")
                break

            # Active trading window
            if self._in_trading_window() and not self.paused:
                tasks = [
                    self._analyse_and_trade(symbol)
                    for symbol in config.WATCHLIST
                ]
                await asyncio.gather(*tasks, return_exceptions=True)
            elif self.paused:
                logger.info("Trading paused.")

            await asyncio.sleep(config.POLL_INTERVAL_SEC)
```

- **Config-drift audit — `python algo.py check-env`.** Settings live in three places that drift apart silently: your gitignored `.env`, the committed `env.example` template, and the in-code default behind every `_env_*` call. A key present in the code and the template but missing from `.env` is **not** an error — the runner just uses the in-code default — which is what makes it easy to miss: an unseen default ends up governing a live-money run. This read-only command reports settings missing from your `.env`, mistyped or stale keys (a typo means the setting you intended is not being applied at all), and knobs missing from the template. It exits non-zero on findings so it can gate a pre-flight script, and prints key **names only** — never a value out of your `.env` — so its output is safe to share. CI enforces the same rule in the other direction: a new `_env_*` key cannot land without its `env.example` entry.
- **Per-strategy size multiplier — `<PREFIX>_SIZE_MULTIPLIER`.** One knob per strategy (default **1**, whole numbers up to **25**) that scales that strategy's entire size/risk set together: its lot count, its per-trade risk budget, its hard lot cap, and its daily max-loss kill-switch. At `2` a 5-lot cap becomes 10, a Rs.2,500 budget becomes Rs.5,000, a Rs.5,500 daily cap becomes Rs.11,000, and a setup that would have taken 4 lots takes 8 — so position size can grow with the **account** by editing one number instead of four that must be kept consistent by hand. Deliberately **per-strategy only** (no global switch, so one typo cannot enlarge every enabled strategy) and it applies to **paper and live alike**, so an enlarged size can be paper-validated first. Anything malformed (`0`, `2.5`, `30`, `"two"`) falls back to 1 for paper and **blocks that strategy from live trading** rather than guessing a size. Leave every multiplier unset to trade exactly as before. Two consequences worth knowing: the scaled budget also loosens the "one lot exceeds the budget" skip, and because lots are floored a 2x can land slightly above a pure doubling (still strictly inside the scaled budget). For SL Hunting, note its BankNIFTY mirror already roughly doubles basket risk, so a multiplier of M leaves the basket near 2xM times the single-leg budget.
