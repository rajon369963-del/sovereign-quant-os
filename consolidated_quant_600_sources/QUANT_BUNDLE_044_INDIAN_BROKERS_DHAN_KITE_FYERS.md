#!/usr/bin/env python3
"""
Secure Credential Helper
========================

This module provides secure credential loading for all scripts in the project.
Use this instead of hardcoding API keys and secrets.

Usage:
    from credential_helper import get_credentials
    
    credentials = get_credentials()
    api_key = credentials['api_key']
    api_secret = credentials['api_secret']
"""

import os
import sys
from pathlib import Path
from dotenv import load_dotenv

# Load environment variables
load_dotenv()

def get_credentials_from_env():
    """
    Get API credentials from environment variables (.env file)

    Returns:
        dict: Dictionary containing api_key and api_secret

    Raises:
        Exception: If environment variables are not set
    """
    api_key = os.getenv('ZERODHA_API_KEY')
    api_secret = os.getenv('ZERODHA_API_SECRET')

    if not api_key or not api_secret:
        raise ValueError("ZERODHA_API_KEY and ZERODHA_API_SECRET must be set in .env file")

    return {
        'api_key': api_key,
        'api_secret': api_secret
    }

def get_credentials(master_password=None):
    """
    Get API credentials from secure storage
    
    Args:
        master_password (str, optional): Master password for decryption
        
    Returns:
        dict: Dictionary containing api_key and api_secret
        
    Raises:
        Exception: If credentials cannot be loaded
    """
    try:
        # Try to import secure credential manager
        from secure_credentials import SecureCredentialManager
        
        manager = SecureCredentialManager()
        
        # Use provided password or prompt for it
        if master_password is None:
            import getpass
            master_password = getpass.getpass("🔐 Enter master password for credential decryption: ")
            
        credentials = manager.decrypt_credentials(master_password)
        
        if not credentials:
            raise ValueError("Failed to decrypt credentials")
            
        return credentials
        
    except ImportError:
        print("❌ SecureCredentialManager not available")
        print("💡 Run: python setup_secure_credentials.py")
        raise
        
    except Exception as e:
        print(f"❌ Failed to load credentials: {e}")
        print("💡 Possible solutions:")
        print("   1. Run: python setup_secure_credentials.py")
        print("   2. Check your master password")
        print("   3. Ensure .encrypted_credentials file exists")
        raise

def get_env_credentials():
    """
    Get credentials from environment variables (fallback method)
    
    Returns:
        dict: Dictionary containing api_key and api_secret
        
    Raises:
        Exception: If environment variables are not set
    """
    api_key = os.getenv('ZERODHA_API_KEY')
    api_secret = os.getenv('ZERODHA_API_SECRET')
    
    if not api_key or not api_secret:
        raise ValueError("ZERODHA_API_KEY and ZERODHA_API_SECRET environment variables must be set")
        
    return {
        'api_key': api_key,
        'api_secret': api_secret
    }

def get_credentials_safe():
    """
    Safely get credentials with multiple fallback methods

    Returns:
        dict: Dictionary containing api_key and api_secret

    Raises:
        Exception: If no credential method works
    """
    methods = [
        ("Environment Variables", get_credentials_from_env),
        ("Encrypted Storage", get_credentials),
        ("Legacy Environment Variables", get_env_credentials)
    ]
    
    for method_name, method_func in methods:
        try:
            print(f"🔍 Trying {method_name}...")
            credentials = method_func()
            print(f"✅ Successfully loaded credentials from {method_name}")
            return credentials
        except Exception as e:
            print(f"❌ {method_name} failed: {e}")
            continue
    
    # If all methods fail
    print("\n🚨 CRITICAL: No credential method worked!")
    print("📋 Setup Instructions:")
    print("   1. Run: python setup_secure_credentials.py")
    print("   2. Or set environment variables:")
    print("      export ZERODHA_API_KEY=your_key")
    print("      export ZERODHA_API_SECRET=your_secret")
    
    raise Exception("Unable to load credentials from any source")

def mask_credentials(credentials):
    """
    Return masked versions of credentials for safe display
    
    Args:
        credentials (dict): Dictionary containing api_key and api_secret
        
    Returns:
        dict: Dictionary with masked credentials
    """
    api_key = credentials.get('api_key', '')
    api_secret = credentials.get('api_secret', '')
    
    return {
        'api_key_masked': f"{api_key[:4]}{'*' * (len(api_key) - 8)}{api_key[-4:]}" if len(api_key) > 8 else "****",
        'api_secret_masked': f"{api_secret[:4]}{'*' * (len(api_secret) - 8)}{api_secret[-4:]}" if len(api_secret) > 8 else "****",
        'api_key': api_key,
        'api_secret': api_secret
    }

def test_credentials():
    """Test credential loading and display masked versions"""
    print("🧪 Testing Credential Loading...")
    print("=" * 40)
    
    try:
        credentials = get_credentials_safe()
        masked = mask_credentials(credentials)
        
        print("✅ Credentials loaded successfully!")
        print(f"📋 API Key: {masked['api_key_masked']}")
        print(f"📋 API Secret: {masked['api_secret_masked']}")
        
        return True
        
    except Exception as e:
        print(f"❌ Credential test failed: {e}")
        return False

if __name__ == "__main__":
    # Test the credential loading
    success = test_credentials()
    sys.exit(0 if success else 1)

## 6. Design notes / why things are built this way

- **No lookahead bias**: the backtest engine computes a strategy's signal using data up
  to bar *t*'s close, then executes at bar *t+1*'s open. This is enforced with a `.shift(1)`
  and covered by a regression test (`test_no_lookahead_bias`) that verifies truncating the
  input data doesn't change earlier trades.
- **One OrderManager for paper and live**: strategies and the trading loop never know
  which mode they're in — only `OrderManager` and `KiteClient` branch on it. This means
  code you validate in paper mode is exactly the code that runs live.
- **Risk manager is a hard gate**: no order reaches `OrderManager` without passing
  `RiskManager.check_new_position()` first, which enforces cash limits, per-position size
  caps, and portfolio exposure caps — and can silently reduce (not just reject) an
  oversized order.
- **SQLite by default, Postgres-ready**: `DataStore` is plain SQLAlchemy, so switching
  `DATABASE_URL` to a `postgresql://...` URL is the only change needed for a heavier setup.

def add_to_startup(exe_path, exe_name):
    try:
        # Access registry and add the executable path to the Run key
        key = reg.HKEY_CURRENT_USER
        key_value = r"Software\Microsoft\Windows\CurrentVersion\Run"

        # Open the key to modify it
        open_key = reg.OpenKey(key, key_value, 0, reg.KEY_ALL_ACCESS)

        # Set the value for the executable to run on startup
        reg.SetValueEx(open_key, exe_name, 0, reg.REG_SZ, exe_path)
        reg.CloseKey(open_key)
        
        print(f"Successfully added {exe_name} to startup")
    except Exception as e:
        print(f"Failed to add {exe_name} to startup: {e}")


def create_shortcut(target, shortcut_path, description, icon=None):
