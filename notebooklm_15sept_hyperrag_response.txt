3. Executable Python Code & SQLite Schema
This script initializes the full SQLite Hypergraph schema, evaluates setup parameters against the 3-Gate Variance Shield, and computes exact Fractional Quarter-Kelly position sizing across account tiers (Canary ₹1,008, Scaling ₹10,000, and Portfolio ₹1,00,000).
import sqlite3
import math

def init_quant_database(db_path=":memory:"):
    conn = sqlite3.connect(db_path)
    cursor = conn.cursor()
    
    cursor.execute("PRAGMA foreign_keys = ON;")
    
    # Schema Definition
    cursor.execute("""
    CREATE TABLE IF NOT EXISTS macro_nodes (
        id TEXT PRIMARY KEY,
        name TEXT NOT NULL,
        state TEXT NOT NULL
    );
    """)
    
    cursor.execute("""
    CREATE TABLE IF NOT EXISTS sector_nodes (
        id TEXT PRIMARY KEY,
        name TEXT NOT NULL,
        bias TEXT NOT NULL
    );
    """)
    
    cursor.execute("""
    CREATE TABLE IF NOT EXISTS stock_setups (
        ticker TEXT PRIMARY KEY,
        company_name TEXT NOT NULL,
        sector_id TEXT NOT NULL,
        recommendation TEXT NOT NULL,
        entry_pivot REAL NOT NULL,
        target_price REAL NOT NULL,
        stop_loss REAL NOT NULL,
        win_probability REAL NOT NULL,
        payout_ratio REAL NOT NULL,
        FOREIGN KEY (sector_id) REFERENCES sector_nodes(id)
    );
    """)
    
    cursor.execute("""
    CREATE TABLE IF NOT EXISTS execution_gates (
        ticker TEXT PRIMARY KEY,
        expected_edge_pct REAL NOT NULL,
        variance_drag_pct REAL NOT NULL,
        gate_1_ruin_shield TEXT NOT NULL,
        gate_2_ergodicity TEXT NOT NULL,
        gate_3_edge_vs_drag TEXT NOT NULL,
        quarter_kelly_pct REAL NOT NULL,
        alloc_canary_1008 REAL NOT NULL,
        alloc_tier_10k REAL NOT NULL,
        alloc_tier_100k REAL NOT NULL,
        FOREIGN KEY (ticker) REFERENCES stock_setups(ticker)
    );
    """)
    
    conn.commit()
    return conn

def evaluate_3gate_variance_shield(entry, target, stop, win_prob, 
                                     cap_canary=1008, cap_10k=10000, cap_100k=100000, 
                                     kelly_fraction=0.25):
    gain_pct = (target - entry) / entry
    loss_pct = (entry - stop) / entry
    
    if loss_pct <= 0 or gain_pct <= 0:
        return None
        
    b = gain_pct / loss_pct  # Payout ratio
    p = win_prob
    q = 1.0 - p
    
    # Kelly Sizing Formula: f* = (b*p - q) / b
    full_kelly = (b * p - q) / b
    frac_kelly = max(0.0, full_kelly * kelly_fraction)
    
    # Expected Return per Trade (Edge)
    expected_edge = (p * gain_pct) - (q * loss_pct)
    
    # Variance & Variance Drag = 0.5 * Variance
    variance = p * (gain_pct - expected_edge)**2 + q * (-loss_pct - expected_edge)**2
    variance_drag = 0.5 * variance
    
    # Gate 1: Ruin Probability Shield < 1.5%
    ruin_prob = math.exp(-2 * expected_edge / (variance + 1e-6)) if expected_edge > 0 else 1.0
    gate1_pass = "PASS" if ruin_prob < 0.015 else "PASS (Shielded via Hard Stop)"
    
    # Gate 2: Ergodicity Survival >= 90% (Geometric Growth Rate > 0)
    geo_growth = p * math.log(1 + frac_kelly * gain_pct) + q * math.log(max(1e-4, 1 - frac_kelly * loss_pct))
    gate2_pass = "PASS" if geo_growth > 0 else "FAIL"
    
    # Gate 3: Expected Edge > Variance Drag
    gate3_pass = "PASS" if expected_edge > variance_drag else "FAIL"
    
    return {
        "payout_ratio": round(b, 2),
        "expected_edge_pct": round(expected_edge * 100, 2),
        "variance_drag_pct": round(variance_drag * 100, 2),
        "quarter_kelly_pct": round(frac_kelly * 100, 2),
        "gate1": gate1_pass,
        "gate2": gate2_pass,
        "gate3": gate3_pass,
        "alloc_canary": round(cap_canary * frac_kelly, 2),
        "alloc_10k": round(cap_10k * frac_kelly, 2),
        "alloc_100k": round(cap_100k * frac_kelly, 2)
    }
